                            balance = balance * 0.80;

                            isDiscounted = true;
                            break;

                            System.out.println("No discount for regular students.");
                            isDiscounted = true;
                            break;

                            System.out.println("Enter appropriate amount.");

                        System.out.println("Balance : " + balance);

                        System.out.println("Balance: No Remaining Balance.");
                    }
                    break;

import java.util.Scanner;

public class BSIT2_4_Cortas_MarkJenel_ATMWithdrawalLimit {

    public static void main(String []args){
         Scanner in = new Scanner(System.in);

        double balance = 9000.0;
        int choice = 0;
        do{
            System.out.println("====ATM Menu====");
            System.out.println("1. Check Balance");
            System.out.println("2. Deposit");
            System.out.println("3. Withdraw");
            System.out.println("4. Exit");

            System.out.println("Enter choice: ");
            choice = in.nextInt();

            switch(choice){
                case 1:
                    System.out.println("Balance: " + balance);
                    break;
                case 2:
                    System.out.println("Enter Deposit Amount: ");
                    double amount = in.nextDouble();
                    balance = balance + amount;
                    break;
                case 3:
                    System.out.println("Enter amount to withdraw: ");
                    double withdraw = in.nextDouble();

                    if(withdraw <= 0) {
                        System.out.println("Transaction Error: Please enter amount bigger than 0.");
                        break;
                    }else if(withdraw > 3000){
                        System.out.println("Transaction Error: Withdrawal Limit Exceeded");
                        break;
                    }else if(withdraw > balance){
                        System.out.println("Transaction Error: Withdrawal Limit Exceeded");
                        break;
                    }else{
                        System.out.println("Withdrawing " + withdraw + " from balance");
                        balance -= withdraw;
                        break;
                    }
                case 4:
                    System.out.println("Exiting the program..");
                    break;
            }
        }while(choice != 4);
    }
}

import java.util.Scanner;

public class ExpenseTrackerCortas {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        // Display program title
        displayTitle();

        // Getting user input
        System.out.print("Enter your total budget: ");
        double budget = in.nextDouble();

        System.out.print("Enter food expenses: ");
        double food = in.nextDouble();

        System.out.print("Enter transportation expenses: ");
        double transport = in.nextDouble();

        System.out.print("Enter extra expenses: ");
        double others = in.nextDouble();

        // Calculate total
        double total = calculateTotalExpenses(food, transport, others);

        // Check budget status
        String status = checkBudgetStatus(total, budget);

        // Final output
        displaySummary(total, status, budget);

        in.close();//close the scanner
    }

    //displays title
    public static void displayTitle() {
        System.out.println("MY PERSONAL EXPENSE TRACKER");
    }

    public static double calculateTotalExpenses(double food, double transport, double others) {
        return food + transport + others;
    }//displays the total value

    //checks the budget status
    public static String checkBudgetStatus(double total, double budget) {
        if (total > budget) {
            return "You have exceeded your budget!";
        } else {
            return "Great job! You still have money left on your budget!";
        }
    }

    //shows total expense and budget status
    public static void displaySummary(double total, String status, double budget) {
        double remaining = budget - total;
        
        System.out.println("\nSummary of Financial Spendings");
        System.out.println("Total Expenses:  P" + total);
        System.out.println("Budget Status:   " + status);
        
        //compute remaining budget
        if (remaining >= 0) {
            System.out.println("Balance: P" + remaining);
            System.out.println("Good financial decisions so far!"); //personalized message
        } else {
            double overspent = Math.abs(remaining);
            System.out.println("Overspent Amount:  P" + overspent);
            System.out.println("Do better next time!"); //personalized message
        }
    }
}
